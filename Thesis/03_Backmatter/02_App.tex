\chapter{Mathematical Definitions}
\label{chap:math_def}



\begin{equation} \label{eq:precision}
\text{Precision} = \frac{TP}{TP + FP}
\end{equation}

\begin{equation} \label{eq:recall}
\text{Recall} = \frac{TP}{TP + FN}
\end{equation}

\begin{equation} \label{eq:val_loss}
\mathcal{L}_{val} = -\frac{1}{N} \sum_{i=1}^{N} [y_i \log(\hat{y}_i) + (1 - y_i) \log(1 - \hat{y}_i)]
\end{equation}

\begin{equation} \label{eq:hamming_loss_xor}
\text{Hamming Loss} = \frac{1}{N \cdot L} \sum_{i=1}^{N} |y_i \oplus \hat{y}_i|
\end{equation}

\begin{equation} \label{eq:jaccard_index}
J(A, B) = \frac{|A \cap B|}{|A \cup B|}
\end{equation}
